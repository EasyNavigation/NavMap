60class OccupancyLayer
61{
62public:
63 static constexpr float kScale = 1000.0f;
64
65 static int16_t to_fixed(float logodds)
66 {
67 const float v = std::round(logodds * kScale);
68 return static_cast<int16_t>(std::clamp(v, -32767.0f, 32767.0f));
69 }
70 static float logodds_of(float p) {return std::log(p / (1.0f - p));}
71 static float probability_of(float logodds) {return 1.0f / (1.0f + std::exp(-logodds));}
72
74 explicit OccupancyLayer(
75 NavMap & nm, const std::string & name = "occupancy_logodds", OccupancyParams p = {})
76 : name_(name),
77 hit_(to_fixed(logodds_of(p.p_hit))), miss_(to_fixed(logodds_of(p.p_miss))),
78 min_(to_fixed(logodds_of(p.p_min))), max_(to_fixed(logodds_of(p.p_max))),
79 occ_(to_fixed(logodds_of(p.p_occupied))), free_(to_fixed(logodds_of(p.p_free)))
80 {
81 const bool existed = nm.has_layer(name);
82 view_ = nm.add_layer<int16_t>(name, "occupancy log-odds", "1/1000 log-odds", 0);
83 if (!existed) {nm.set_layer_unknown(name, 0.0);}
84 }
85
87 void hit(NavCelId cid) {add(cid, hit_);}
89 void miss(NavCelId cid) {add(cid, miss_);}
90 void update(NavCelId cid, bool occupied) {add(cid, occupied ? hit_ : miss_);}
91
93 void add(NavCelId cid, int16_t delta)
94 {
95 auto & d = view_->data();
96 if (cid >= d.size()) {return;}
97 const int v = static_cast<int>(d[cid]) + delta;
98 d[cid] = static_cast<int16_t>(std::clamp(v, static_cast<int>(min_), static_cast<int>(max_)));
99 view_->mark_changed(cid);
100 }
101
103 void reset(NavCelId cid)
104 {
105 auto & d = view_->data();
106 if (cid < d.size()) {d[cid] = 0; view_->mark_changed(cid);}
107 }
108
109 float logodds(NavCelId cid) const {return value(cid) / kScale;}
110 float probability(NavCelId cid) const {return probability_of(logodds(cid));}
111
112 OccupancyState state(NavCelId cid) const
113 {
114 const int16_t v = value(cid);
115 if (v >= occ_) {return OccupancyState::OCCUPIED;}
116 if (v <= free_) {return OccupancyState::FREE;}
117 return OccupancyState::UNKNOWN;
118 }
119
120 const std::string & name() const {return name_;}
121 const std::shared_ptr<LayerView<int16_t>> & view() const {return view_;}
122
123private:
124 int16_t value(NavCelId cid) const
125 {
126 const auto & d = view_->data();
127 return cid < d.size() ? d[cid] : int16_t{0};
128 }
129
130 std::string name_;
131 std::shared_ptr<LayerView<int16_t>> view_;
132 int16_t hit_, miss_, min_, max_, occ_, free_;
133};
