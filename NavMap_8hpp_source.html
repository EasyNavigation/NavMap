124struct LayerViewBase
125{
126 virtual ~LayerViewBase() = default;
127
129 virtual LayerType type() const = 0;
130
132 virtual const std::string & name() const = 0;
133
135 virtual size_t size() const = 0;
136
138 virtual std::uint64_t content_hash() const = 0;
139
141 void mark_dirty() const noexcept {hash_dirty_ = true;}
142
144 virtual std::shared_ptr<LayerViewBase> clone() const = 0;
145
146 // ---- Change tracking (for incremental consumers: inflation, publication...) ----
147 // Cells written since the last clear_changes(): recorded by NavMap::layer_set(), OccupancyLayer
148 // and mark_changed(). Bulk writes (set_data, layer_clear, copies) mark the whole layer, and so
149 // does creating it. Direct writes through data() must call mark_changed() themselves.
150
152 void mark_changed(NavCelId cid) const
153 {
154 hash_dirty_ = true;
155 if (all_changed_) {return;}
156 const std::size_t word = cid / 64;
157 if (changed_bits_.size() <= word) {
158 changed_bits_.resize(std::max(word + 1, (size() + 63) / 64));
159 }
160 const std::uint64_t bit = std::uint64_t{1} << (cid % 64);
161 if (!(changed_bits_[word] & bit)) {
162 changed_bits_[word] |= bit;
163 changed_.push_back(cid);
164 }
165 }
166
168 void mark_all_changed() const noexcept {hash_dirty_ = true; all_changed_ = true;}
169
171 bool all_changed() const noexcept {return all_changed_;}
172
174 const std::vector<NavCelId> & changed() const noexcept {return changed_;}
175
176 bool has_changes() const noexcept {return all_changed_ || !changed_.empty();}
177
179 void clear_changes() const
180 {
181 for (const NavCelId c : changed_) {changed_bits_[c / 64] &= ~(std::uint64_t{1} << (c % 64));}
182 changed_.clear();
183 all_changed_ = false;
184 }
185
186protected:
187 mutable bool hash_dirty_{true};
188 mutable std::uint64_t hash_cache_{0};
189 mutable bool all_changed_{true}; // a new layer is all new
190 mutable std::vector<NavCelId> changed_;
191 mutable std::vector<std::uint64_t> changed_bits_; // allocated on the first single change
192};

248namespace detail

250inline std::uint64_t fnv1a64_bytes(
251 const void * data, std::size_t n,
252 std::uint64_t seed = 1469598103934665603ULL)
253{
254 const auto * p = static_cast<const std::uint8_t *>(data);
255 std::uint64_t h = seed;
256 for (std::size_t i = 0; i < n; ++i) {
257 h ^= p[i]; h *= 1099511628211ULL;
258 }
259 return h;
260}
261} // namespace detail

264template<typename T>
265std::uint64_t LayerView<T>::content_hash() const
266{
267 if (!hash_dirty_) {return hash_cache_;}
268 const std::size_t n = data_.size();
269 std::uint64_t h = navmap::detail::fnv1a64_bytes(&n, sizeof(n));
270 if (n) {
271 static_assert(
272 std::is_trivially_copyable<T>::value,
273 "LayerView<T> requires trivially copyable T.");
274 h = navmap::detail::fnv1a64_bytes(data_.data(), n * sizeof(T), h);
275 }
276 hash_cache_ = h;
277 hash_dirty_ = false;
278 return hash_cache_;
279}

937 template<typename T>
938 T layer_get(const std::string & name, NavCelId cid, T def = T{}) const
939 {
940 if (auto base = layers.get(name)) {
941 if (auto view = std::dynamic_pointer_cast<LayerView<T>>(base)) {
942 const auto & buf = view->data();
943 if (static_cast<size_t>(cid) < buf.size()) {
944 return buf[cid];
945 }
946 return def;
947 }
948 }
949 // Fallback: try to fetch as double from any stored type and convert
950 double v;
951 if (auto base = layers.get(name)) {
952 if (auto v8 = std::dynamic_pointer_cast<LayerView<uint8_t>>(base)) {
953 const auto & buf = v8->data();
954 if (static_cast<size_t>(cid) >= buf.size()) {return def;}
955 v = static_cast<double>(buf[cid]);
956 } else if (auto vf = std::dynamic_pointer_cast<LayerView<float>>(base)) {
957 const auto & buf = vf->data();
958 if (static_cast<size_t>(cid) >= buf.size()) {return def;}
959 v = static_cast<double>(buf[cid]);
960 } else if (auto vd = std::dynamic_pointer_cast<LayerView<double>>(base)) {
961 const auto & buf = vd->data();
962 if (static_cast<size_t>(cid) >= buf.size()) {return def;}
963 v = buf[cid];
964 } else if (auto vi = std::dynamic_pointer_cast<LayerView<int16_t>>(base)) {
965 const auto & buf = vi->data();
966 if (static_cast<size_t>(cid) >= buf.size()) {return def;}
967 v = static_cast<double>(buf[cid]);
968 } else {
969 return def;
970 }
971 } else {
972 return def;
973 }
974
975 if constexpr (std::is_floating_point_v<T>) {
976 return static_cast<T>(v);
977 } else if constexpr (std::is_integral_v<T>) {
978 const double lo = static_cast<double>(std::numeric_limits<T>::lowest());
979 const double hi = static_cast<double>(std::numeric_limits<T>::max());
980 double vv = v < lo ? lo : (v > hi ? hi : v);
981 return static_cast<T>(std::llround(vv));
982 } else {
983 static_assert(sizeof(T) == 0, "Unsupported layer type");
984 }
985 }

1048 template<typename T>
1049 bool layer_copy(const std::string & src, const std::string & dst)
1050 {
1051 // No-op if names are equal
1052 if (src == dst) {
1053 return true;
1054 }
1055
1056 // Get source
1057 auto src_base = layers.get(src);
1058 auto src_view = std::dynamic_pointer_cast<LayerView<T>>(src_base);
1059 if (!src_view) {
1060 return false; // source layer missing or wrong type
1061 }
1062
1063 // Ensure destination exists with correct length (navcels.size())
1064 auto dst_view = layers.add_or_get<T>(dst, navcels.size(), layer_type_tag<T>());
1065 if (!dst_view) {
1066 return false;
1067 }
1068
1069 // If sizes mismatch, something is wrong with geometry/layer size.
1070 // You can choose to resize here if your semantics allow it. For safety, fail:
1071 if (src_view->data().size() != dst_view->data().size()) {
1072 return false;
1073 }
1074
1075 // If both views already refer to the same underlying buffer, no work to do
1076 if (!src_view->data().empty() &&
1077 src_view->data().data() == dst_view->data().data())
1078 {
1079 return true;
1080 }
1081
1082 // Hash-based skip: O(1) if cached, O(n) only on first compute (then cached)
1083 const bool same_hash = (src_view->content_hash() == dst_view->content_hash());
1084 if (same_hash) {
1085 return true; // identical content → avoid copy
1086 }
1087
1088 // Copy only when different. set_data() marks hash as dirty for dst.
1089 dst_view->set_data(src_view->data());
1090 return true;
1091 }

1109 template<typename T>
1110 std::shared_ptr<LayerView<T>> add_layer(
1111 const std::string & name,
1112 const std::string & description,
1113 const std::string & unit,
1114 const std::string & src_layer)
1115 {
1116 auto dst = layers.add_or_get<T>(name, navcels.size(), layer_type_tag<T>());
1117 if (dst->data().size() != navcels.size()) {
1118 dst->data().assign(navcels.size(), T{});
1119 }
1120 auto & meta = layer_meta[name];
1121 meta.description = description;
1122 meta.unit = unit;
1123 meta.per_cell = true;
1124
1125 if (!has_layer(src_layer)) {
1126 return dst;
1127 }
1128
1129 if (layer_copy<T>(src_layer, name)) {
1130 return dst;
1131 }
1132
1133 const std::size_t n = dst->size();
1134 for (NavCelId cid = 0; cid < n; ++cid) {
1135 double v = layer_get<double>(src_layer, cid, std::numeric_limits<double>::quiet_NaN());
1136 if (std::isnan(v)) {v = 0.0;}
1137
1138 if constexpr (std::is_floating_point_v<T>) {
1139 dst->data()[cid] = static_cast<T>(v);
1140 } else if constexpr (std::is_integral_v<T>) {
1141 const double lo = static_cast<double>(std::numeric_limits<T>::lowest());
1142 const double hi = static_cast<double>(std::numeric_limits<T>::max());
1143 if (v < lo) {v = lo;}
1144 if (v > hi) {v = hi;}
1145 dst->data()[cid] = static_cast<T>(std::llround(v));
1146 } else {
1147 static_assert(sizeof(T) == 0, "Unsupported layer type");
1148 }
1149 }
1150
1151 return dst;
1152 }

1193 if (navcels.empty() || surfaces.empty() || size <= 0.0f) {
1194 return false;
1195 }
1196
1197 std::size_t surface_idx = 0;
1198 NavCelId start_cid{};
1199 Eigen::Vector3f bary{};
1200 Eigen::Vector3f hit_pt{};
1201 if (!locate_navcel(p_world, surface_idx, start_cid, bary, &hit_pt)) {
1202 return false;
1203 }
1204
1205 const float cx = hit_pt.x();
1206 const float cy = hit_pt.y();
1207
1208 {
1209 auto existing = layers.get(layer_name);
1210 if (existing && existing->type() != layer_type_tag<T>()) {
1211 return false; // type mismatch
1212 }
1213 }
1214 auto layer = layers.add_or_get<T>(layer_name, navcels.size(), layer_type_tag<T>());
1215 if (!layer) {
1216 return false;
1217 }
1218 if (layer->data().size() != navcels.size()) {
1219 const_cast<std::vector<T> &>(layer->data()).resize(navcels.size(), T{});
1220 }
1221
1222 const float half = (shape == AreaShape::CIRCULAR) ? size : (0.5f * size);
1223 const float minx = cx - half, maxx = cx + half;
1224 const float miny = cy - half, maxy = cy + half;
1225 const float r2 = (shape == AreaShape::CIRCULAR) ? (size * size) : 0.0f;
1226
1227 auto tri_aabb_intersects_box = [&](NavCelId cid) -> bool {
1228 const auto & tri = navcels[static_cast<std::size_t>(cid)];
1229 const Eigen::Vector3f p0{positions.x[tri.v[0]], positions.y[tri.v[0]],
1230 positions.z[tri.v[0]]};
1231 const Eigen::Vector3f p1{positions.x[tri.v[1]], positions.y[tri.v[1]],
1232 positions.z[tri.v[1]]};
1233 const Eigen::Vector3f p2{positions.x[tri.v[2]], positions.y[tri.v[2]],
1234 positions.z[tri.v[2]]};
1235 const float tminx = std::min({p0.x(), p1.x(), p2.x()});
1236 const float tmaxx = std::max({p0.x(), p1.x(), p2.x()});
1237 const float tminy = std::min({p0.y(), p1.y(), p2.y()});
1238 const float tmaxy = std::max({p0.y(), p1.y(), p2.y()});
1239 if (tmaxx < minx || tminx > maxx || tmaxy < miny || tminy > maxy) {
1240 return false;
1241 }
1242 return true;
1243 };
1244
1245 auto inside_area = [&](const Eigen::Vector3f & c) -> bool {
1246 const float dx = c.x() - cx;
1247 const float dy = c.y() - cy;
1248 if (shape == AreaShape::CIRCULAR) {
1249 return (dx * dx + dy * dy) <= r2;
1250 } else {
1251 return (std::abs(dx) <= half) && (std::abs(dy) <= half);
1252 }
1253 };
1254
1255 std::vector<char> visited(navcels.size(), 0);
1256 std::deque<NavCelId> q;
1257 q.push_back(start_cid);
1258 visited[static_cast<std::size_t>(start_cid)] = 1;
1259
1260 auto & data = layer->mutable_data();
1261
1262 while (!q.empty()) {
1263 const NavCelId cid = q.front(); q.pop_front();
1264
1265 if (!tri_aabb_intersects_box(cid)) {
1266 continue;
1267 }
1268
1269 const Eigen::Vector3f c = navcel_centroid(cid);
1270 if (inside_area(c)) {
1271 data[static_cast<std::size_t>(cid)] = value;
1272 }
1273
1274 for (const auto ncid : navcel_neighbors(cid)) {
1275 const std::size_t nidx = static_cast<std::size_t>(ncid);
1276 if (nidx < navcels.size() && !visited[nidx] && tri_aabb_intersects_box(ncid)) {
1277 visited[nidx] = 1;
1278 q.push_back(ncid);
1279 }
1280 }
1281 }
1282
1283 return true;
1284 }
